    71→    71→def _draw_hud(frame, gesture_result, spring, fps, config,
    72→    72→              frozen, dragging, scrolling, volume_mode, brightness_mode,
    73→    73→              precision_mode, gsm_progress, gesture_confidence):
    74→    74→    """Draw Iron Man HUD overlay with v3.1 enhancements."""
    75→    75→    if frame is None:
    76→    76→        return
    77→    77→    h, w = frame.shape[:2]
    78→    78→    gesture = gesture_result.get("gesture", "none")
    79→    79→
    80→    80→    colors = {
    81→    81→        Gesture.POINTING: (0, 255, 255), Gesture.PINCH: (0, 255, 0),
    82→    82→        Gesture.PEACE: (0, 255, 200), Gesture.PALM: (255, 128, 0),
    83→    83→        Gesture.FIST: (0, 0, 255), Gesture.THUMBS_UP: (0, 200, 0),
    84→    84→        Gesture.THREE: (255, 0, 255), Gesture.PINKY: (200, 200, 0),
    85→    85→        Gesture.GUN: (0, 200, 200), Gesture.ROCK: (200, 0, 200),
    86→    86→        Gesture.SHAKA: (0, 200, 100), Gesture.OK: (0, 215, 255),
    87→    87→        Gesture.RING: (255, 200, 0), Gesture.SIX: (100, 200, 200),
    88→    88→        "none": (80, 80, 80),
    89→    89→    }
    90→    90→    color = colors.get(gesture, (80, 80, 80))
    91→    91→
    92→    92→    # Gesture indicator (top-left) with confirm progress ring
    93→    93→    cx, cy = 50, 50
    94→    94→    # Outer ring — progress indicator
    95→    95→    if gsm_progress > 0 and gsm_progress < 1.0:
    96→    96→        # Draw progress arc
    97→    97→        end_angle = int(360 * gsm_progress)
    98→    98→        cv2.ellipse(frame, (cx, cy), (30, 30), -90, 0, end_angle, (0, 255, 255), 3)
    99→    99→        cv2.circle(frame, (cx, cy), 30, (40, 40, 40), 1)
   100→   100→    else:
   101→   101→        cv2.circle(frame, (cx, cy), 28, color, 3)
   102→   102→    cv2.circle(frame, (cx, cy), 4, color, -1)
   103→   103→
   104→   104→    info = GESTURE_INFO.get(gesture)
   105→   105→    label = info["name"] if info else gesture
   106→   106→    cv2.putText(frame, label.upper(), (cx + 38, cy + 8),
   107→   107→                cv2.FONT_HERSHEY_SIMPLEX, 0.65, color, 2, cv2.LINE_AA)
   108→   108→
   109→   109→    # Confidence bar (next to gesture name)
   110→   110→    if gesture_confidence > 0:
   111→   111→        bar_x = cx + 38 + len(label) * 12 + 10
   112→   112→        bar_w = 40
   113→   113→        cv2.rectangle(frame, (bar_x, cy - 6), (bar_x + bar_w, cy + 6), (40, 40, 40), -1)
   114→   114→        fill_w = int(bar_w * gesture_confidence)
   115→   115→        conf_color = (0, 255, 0) if gesture_confidence > 0.7 else (0, 200, 255)
   116→   116→        cv2.rectangle(frame, (bar_x, cy - 6), (bar_x + fill_w, cy + 6), conf_color, -1)
   117→   117→
   118→   118→    # Status badges (bottom)
   119→   119→    badges = []
   120→   120→    if frozen:
   121→   121→        badges.append(("FROZEN", (0, 0, 255)))
   122→   122→    if dragging:
   123→   123→        badges.append(("DRAG", (255, 128, 0)))
   124→   124→    if scrolling:
   125→   125→        badges.append(("SCROLL", (255, 0, 255)))
   126→   126→    if volume_mode:
   127→   127→        badges.append(("VOL", (0, 200, 100)))
   128→   128→    if brightness_mode:
   129→   129→        badges.append(("BRIGHT", (255, 200, 0)))
   130→   130→    if precision_mode:
   131→   131→        badges.append(("PRECISE", (200, 200, 255)))
   132→   132→    if spring is None:
   133→   133→        badges.append(("DIRECT", (0, 255, 255)))
   134→   134→    bx = 10
   135→   135→    for bt, bc in badges:
   136→   136→        # Badge background
   137→   137→        (tw, _), _ = cv2.getTextSize(bt, cv2.FONT_HERSHEY_SIMPLEX, 0.55, 2)
   138→   138→        cv2.rectangle(frame, (bx - 3, h - 38), (bx + tw + 5, h - 12), (20, 20, 20), -1)
   139→   139→        cv2.putText(frame, bt, (bx, h - 20),
   140→   140→                    cv2.FONT_HERSHEY_SIMPLEX, 0.55, bc, 2, cv2.LINE_AA)
   141→   141→        bx += tw + 18
   142→   142→
   143→   143→    # Physics stats (bottom-right)
   144→   144→    if spring is not None:
   145→   145→        k = spring.current_stiffness
   146→   146→        speed = np.linalg.norm(spring.velocity)
   147→   147→        stats = [f"FPS:{fps:.0f}", f"k:{k:.0f}", f"v:{speed:.0f}"]
   148→   148→    else:
   149→   149→        stats = [f"FPS:{fps:.0f}", "DIRECT", ""]
   150→   150→    for i, t in enumerate(stats):
   151→   151→        if t:
   152→   152→            cv2.putText(frame, t, (w - 120, h - 55 + i * 20),
   153→   153→                        cv2.FONT_HERSHEY_SIMPLEX, 0.45, (0, 255, 0), 1, cv2.LINE_AA)
   154→   154→
   155→   155→    # Draw tracked finger landmarks
   156→   156→    lm = gesture_result.get("landmarks")
   157→   157→    if lm is not None:
   158→   158→        from .gestures import INDEX_TIP, THUMB_TIP
   159→   159→        tip = lm[INDEX_TIP]
   160→   160→        tx, ty = int(tip.x * w), int(tip.y * h)
   161→   161→
   162→   162→        # Cursor crosshair with glow
   163→   163→        glow_color = (0, 80, 80) if not precision_mode else (80, 80, 0)
   164→   164→        cv2.circle(frame, (tx, ty), 11, glow_color, -1)  # Glow
   165→   165→        cv2.circle(frame, (tx, ty), 7, (0, 255, 255) if not precision_mode else (255, 255, 0), -1)
   166→   166→        cv2.circle(frame, (tx, ty), 13, (0, 255, 255) if not precision_mode else (255, 255, 0), 2)
   167→   167→
   168→   168→        thumb = lm[THUMB_TIP]
   169→   169→        thx, thy = int(thumb.x * w), int(thumb.y * h)
   170→   170→        cv2.circle(frame, (thx, thy), 4, (255, 200, 0), -1)
   171→   171→        if gesture == Gesture.PINCH:
   172→   172→            cv2.line(frame, (tx, ty), (thx, thy), (0, 255, 0), 2)

   193→   193→def _safe_kb_action(kb_instance, action_name):
   194→   194→    """Safely call a keyboard action — never crash the main loop."""
   195→   195→    if kb_instance is None:
   196→   196→        return
   197→   197→    try:
   198→   198→        method = getattr(kb_instance, action_name)
   199→   199→        method()
   200→   200→    except Exception:
   201→   201→        pass
