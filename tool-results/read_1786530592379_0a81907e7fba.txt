    50→def _get_screen_size():
    51→    try:
    52→        import ctypes
    53→        user32 = ctypes.windll.user32
    54→        w, h = user32.GetSystemMetrics(0), user32.GetSystemMetrics(1)
    55→        if w > 0 and h > 0:
    56→            return w, h
    57→    except Exception:
    58→        pass
    59→    try:
    60→        import tkinter as tk
    61→        root = tk.Tk()
    62→        w, h = root.winfo_screenwidth(), root.winfo_screenheight()
    63→        root.destroy()
    64→        if w > 0 and h > 0:
    65→            return w, h
    66→    except Exception:
    67→        pass
    68→    return 1920, 1080

    71→def _draw_hud(frame, gesture_result, spring, fps, config,
    72→              frozen, dragging, scrolling, volume_mode, brightness_mode,
    73→              precision_mode, gsm_progress, gesture_confidence):
    74→    """Draw Iron Man HUD overlay with v3.1 enhancements."""
    75→    if frame is None:
    76→        return
    77→    h, w = frame.shape[:2]
    78→    gesture = gesture_result.get("gesture", "none")
    79→
    80→    colors = {
    81→        Gesture.POINTING: (0, 255, 255), Gesture.PINCH: (0, 255, 0),
    82→        Gesture.PEACE: (0, 255, 200), Gesture.PALM: (255, 128, 0),
    83→        Gesture.FIST: (0, 0, 255), Gesture.THUMBS_UP: (0, 200, 0),
    84→        Gesture.THREE: (255, 0, 255), Gesture.PINKY: (200, 200, 0),
    85→        Gesture.GUN: (0, 200, 200), Gesture.ROCK: (200, 0, 200),
    86→        Gesture.SHAKA: (0, 200, 100), Gesture.OK: (0, 215, 255),
    87→        Gesture.RING: (255, 200, 0), Gesture.SIX: (100, 200, 200),
    88→        "none": (80, 80, 80),
    89→    }
    90→    color = colors.get(gesture, (80, 80, 80))
    91→
    92→    # Gesture indicator (top-left) with confirm progress ring
    93→    cx, cy = 50, 50
    94→    # Outer ring — progress indicator
    95→    if gsm_progress > 0 and gsm_progress < 1.0:
    96→        # Draw progress arc
    97→        end_angle = int(360 * gsm_progress)
    98→        cv2.ellipse(frame, (cx, cy), (30, 30), -90, 0, end_angle, (0, 255, 255), 3)
    99→        cv2.circle(frame, (cx, cy), 30, (40, 40, 40), 1)
   100→    else:
   101→        cv2.circle(frame, (cx, cy), 28, color, 3)
   102→    cv2.circle(frame, (cx, cy), 4, color, -1)
   103→
   104→    info = GESTURE_INFO.get(gesture)
   105→    label = info["name"] if info else gesture
   106→    cv2.putText(frame, label.upper(), (cx + 38, cy + 8),
   107→                cv2.FONT_HERSHEY_SIMPLEX, 0.65, color, 2, cv2.LINE_AA)
   108→
   109→    # Confidence bar (next to gesture name)
   110→    if gesture_confidence > 0:
   111→        bar_x = cx + 38 + len(label) * 12 + 10
   112→        bar_w = 40
   113→        cv2.rectangle(frame, (bar_x, cy - 6), (bar_x + bar_w, cy + 6), (40, 40, 40), -1)
   114→        fill_w = int(bar_w * gesture_confidence)
   115→        conf_color = (0, 255, 0) if gesture_confidence > 0.7 else (0, 200, 255)
   116→        cv2.rectangle(frame, (bar_x, cy - 6), (bar_x + fill_w, cy + 6), conf_color, -1)
   117→
   118→    # Status badges (bottom)
   119→    badges = []
   120→    if frozen:
   121→        badges.append(("FROZEN", (0, 0, 255)))
   122→    if dragging:
   123→        badges.append(("DRAG", (255, 128, 0)))
   124→    if scrolling:
   125→        badges.append(("SCROLL", (255, 0, 255)))
   126→    if volume_mode:
   127→        badges.append(("VOL", (0, 200, 100)))
   128→    if brightness_mode:
   129→        badges.append(("BRIGHT", (255, 200, 0)))
   130→    if precision_mode:
   131→        badges.append(("PRECISE", (200, 200, 255)))
   132→    if spring is None:
   133→        badges.append(("DIRECT", (0, 255, 255)))
   134→    bx = 10
   135→    for bt, bc in badges:
   136→        # Badge background
   137→        (tw, _), _ = cv2.getTextSize(bt, cv2.FONT_HERSHEY_SIMPLEX, 0.55, 2)
   138→        cv2.rectangle(frame, (bx - 3, h - 38), (bx + tw + 5, h - 12), (20, 20, 20), -1)
   139→        cv2.putText(frame, bt, (bx, h - 20),
   140→                    cv2.FONT_HERSHEY_SIMPLEX, 0.55, bc, 2, cv2.LINE_AA)
   141→        bx += tw + 18
   142→
   143→    # Physics stats (bottom-right)
   144→    if spring is not None:
   145→        k = spring.current_stiffness
   146→        speed = np.linalg.norm(spring.velocity)
   147→        stats = [f"FPS:{fps:.0f}", f"k:{k:.0f}", f"v:{speed:.0f}"]
   148→    else:
   149→        stats = [f"FPS:{fps:.0f}", "DIRECT", ""]
   150→    for i, t in enumerate(stats):
   151→        if t:
   152→            cv2.putText(frame, t, (w - 120, h - 55 + i * 20),
   153→                        cv2.FONT_HERSHEY_SIMPLEX, 0.45, (0, 255, 0), 1, cv2.LINE_AA)
   154→
   155→    # Draw tracked finger landmarks
   156→    lm = gesture_result.get("landmarks")
   157→    if lm is not None:
   158→        from .gestures import INDEX_TIP, THUMB_TIP
   159→        tip = lm[INDEX_TIP]
   160→        tx, ty = int(tip.x * w), int(tip.y * h)
   161→
   162→        # Cursor crosshair with glow
   163→        glow_color = (0, 80, 80) if not precision_mode else (80, 80, 0)
   164→        cv2.circle(frame, (tx, ty), 11, glow_color, -1)  # Glow
   165→        cv2.circle(frame, (tx, ty), 7, (0, 255, 255) if not precision_mode else (255, 255, 0), -1)
   166→        cv2.circle(frame, (tx, ty), 13, (0, 255, 255) if not precision_mode else (255, 255, 0), 2)
   167→
   168→        thumb = lm[THUMB_TIP]
   169→        thx, thy = int(thumb.x * w), int(thumb.y * h)
   170→        cv2.circle(frame, (thx, thy), 4, (255, 200, 0), -1)
   171→        if gesture == Gesture.PINCH:
   172→            cv2.line(frame, (tx, ty), (thx, thy), (0, 255, 0), 2)

   193→def _safe_kb_action(kb_instance, action_name):
   194→    """Safely call a keyboard action — never crash the main loop."""
   195→    if kb_instance is None:
   196→        return
   197→    try:
   198→        method = getattr(kb_instance, action_name)
   199→        method()
   200→    except Exception:
   201→        pass

   263→    if args.scale is not None: config.exp_scale = args.scale

   272→        selected_display = get_primary_display()

   295→
   296→    # ═══ Physics stack v3.2 ═══
   297→    is_direct = (config.tracking_mode == "direct")
   298→
   299→    # Direct tracker (v3.2 default) — 1:1 finger-to-screen
   300→    direct_tracker = None
   301→    # Ironman components (legacy)
   302→    spring = None
   303→    jitter_x = jitter_y = None
   304→    home = None
   305→    exp_curve = precision_curve = None
   306→    momentum = None
   307→    edge_grav = None
   308→    velocity_predictor = None
   309→    position_smoother = None
   310→

   312→        direct_tracker = DirectTracker(
   313→            screen_w=screen_w, screen_h=screen_h,
   314→            jitter_alpha=config.direct_jitter_alpha,
   315→            spring_alpha=config.direct_spring_alpha,
   316→            smooth_alpha=config.direct_smooth_alpha,
   317→            movement_threshold=config.direct_movement_threshold,
   318→            pixel_deadzone=config.direct_pixel_deadzone,
   319→            mirror_x=config.direct_mirror_x,
   320→        )
   321→    else:
   322→        # Ironman mode physics (legacy v3.1)
   323→        spring = AdaptiveSpringDamper(
   324→            mass=config.mass, stiffness_min=config.stiffness_min,
   325→            stiffness_max=config.stiffness_max, damping_ratio=config.damping_ratio,
   326→            speed_threshold=config.speed_threshold,
   327→            max_accel=config.max_accel,
   328→            stiffness_smoothing=config.stiffness_smoothing,
   329→        )
   330→        # Dual-stage jitter filter (micro-tremor + macro smooth)
   331→        jitter_x = DualStageJitterFilter(
   332→            micro_alpha=config.jitter_micro_alpha,
   333→            macro_alpha=config.jitter_macro_alpha,
   334→        )
   335→        jitter_y = DualStageJitterFilter(
   336→            micro_alpha=config.jitter_micro_alpha,
   337→            macro_alpha=config.jitter_macro_alpha,
   338→        )
   339→        home = HomePosition(
   340→            drift_rate=config.home_drift_rate,
   341→            drift_rate_moving=config.home_drift_rate_moving,
   342→        )
   343→        exp_curve = ExponentialCurve(power=config.exp_power, scale=config.exp_scale)
   344→        precision_curve = ExponentialCurve(power=config.precision_power, scale=config.precision_scale)
   345→        momentum = MomentumThrow(
   346→            friction=config.throw_friction,
   347→            min_speed=config.throw_min_speed,
   348→            max_momentum=config.throw_max_momentum,
   349→        )
   350→        edge_grav = EdgeGravity(
   351→            strength=config.edge_gravity_strength,
   352→            edge_zone=config.edge_gravity_zone,
   353→        )
   354→        # v3.1 new physics components
   355→        velocity_predictor = VelocityPredictor(
   356→            prediction_factor=config.prediction_factor,
   357→            max_correction=config.prediction_max_correction,
   358→        )
   359→        position_smoother = PositionSmoother(alpha=config.position_smooth_alpha)
   360→
   361→    swipe = SwipeDetector()
   362→    gsm = GestureStateMachine(
   363→        confirm_frames=config.gesture_confirm_frames,
   364→        action_confirm_frames=config.gesture_action_confirm_frames,
   365→        transition_cooldown=config.gesture_transition_cooldown,
   366→    )
   367→
   368→    # State
   369→    center = np.array([screen_w / 2.0, screen_h / 2.0])
   370→    if is_direct:
   371→        direct_tracker.reset(center)
   372→    else:
   373→        spring.reset(center)
   374→        position_smoother.reset(center)
   375→    mouse.move_to(center[0], center[1])
   376→
   377→    last_click_time = 0.0
   378→    prev_gesture = Gesture.NONE
   379→    cursor_frozen = False
   380→    dragging = False
   381→    scrolling = False
   382→    volume_mode = False
   383→    brightness_mode = False
   384→    precision_mode = args.precision or config.precision_mode
   385→    scroll_accum = 0.0
   386→    prev_index_y = None
   387→    prev_pos = None
   388→    fps = 0.0
   389→    frame_times = []
   390→    debug_mode = True
   391→    running = True
   392→    gesture_confidence = 0.0
   393→    hand_absent_frames = 0  # Track how long hand has been absent
   394→    hand_was_lost = False    # Track hand-loss for sound feedback
   395→
   396→    # Volume/brightness mode debounce
   397→    last_volume_time = 0.0
   398→    last_brightness_time = 0.0
   399→    VOLUME_COOLDOWN = 0.15   # Seconds between volume adjustments
   400→    BRIGHTNESS_COOLDOWN = 0.3  # Brightness changes are slower
   401→
   402→    # Keyboard actions (lazy-init, never crash)
   403→    kb = None
   404→    def _kb():
   405→        nonlocal kb
   406→        if kb is None:
   407→            try:
   408→                from .keyboard import KeyboardActions
   409→                kb = KeyboardActions()
   410→            except Exception:
   411→                pass
   412→        return kb
   413→
   414→    # Banner
   415→    print()
   416→    print("  +==================================================+")
   417→    print(f"  |     AirMouse v{_pkg.__version__} - Iron Man Next-Gen Edition  |")
   418→    print("  |     14 gestures | Physics cursor | Finger-relative  |")
   419→    print("  +==================================================+")
   420→    print(f"  Screen: {screen_w}x{screen_h}  |  Audio: {'ON' if config.audio_enabled else 'OFF'}  |  Gestures: 14+2 swipe")
   421→    print(f"  Tracking: {'DIRECT (1:1 finger-to-screen)' if is_direct else 'IRONMAN (exponential finger-relative)'}")
   422→    if precision_mode:
   423→        print(f"  Mode: PRECISION (power={config.precision_power}, scale={config.precision_scale})")
   424→    _show_quick_reference()
   425→    print()
   426→
   427→    try:
   428→        while running:
   429→            t0 = time.perf_counter()
   430→            dt = 1.0 / max(fps, 1.0)
   431→            now = time.perf_counter()
   432→
   433→            hand_data = tracker.read()
   434→            gesture_result = {"gesture": Gesture.NONE, "landmarks": None, "confidence": 0.0}
   435→
   436→            if hand_data["hand_found"] and hand_data["landmarks"] is not None:
   437→                hand_absent_frames = 0  # Reset absent counter
   438→                if hand_was_lost:
   439→                    audio.hand_found()
   440→                    hand_was_lost = False
   441→                gesture_result = recognize_gesture(hand_data["landmarks"],
   442→                                                    pinch_threshold=config.pinch_threshold)
   443→                raw_gesture = gesture_result["gesture"]
   444→                gesture_confidence = gesture_result.get("confidence", 0.0)
   445→
   446→                # Gesture state machine — with stability and time
   447→                if is_direct:
   448→                    hand_stable = True  # Direct mode doesn't need stability gating
   449→                else:
   450→                    hand_stable = home.is_stable() if home.is_calibrated else False
   451→                gesture = gsm.update(raw_gesture, now=now, hand_stable=hand_stable)
   452→                gesture_changed = (gesture != prev_gesture)
   453→
   454→                # Sound feedback on gesture confirmation
   455→                if gesture_changed and gesture != Gesture.NONE:
   456→                    if prev_gesture == Gesture.NONE or gsm.progress >= 1.0:
   457→                        audio.gesture_confirm()
   458→
   459→                raw_pos = gesture_result["index_pos"]
   460→
   461→                if is_direct:
   462→                    # ═══ DIRECT TRACKING (v3.2) ═══
   463→                    # 1:1 finger-to-screen — no home, no delta, no drift
   464→                    cursor_pos = direct_tracker.update(raw_pos, dt)
   465→                    if not cursor_frozen:
   466→                        mouse.move_to(cursor_pos[0], cursor_pos[1])
   467→                    speed = np.linalg.norm(direct_tracker.velocity)
   468→                    # In direct mode, filtered_pos = raw_pos (no multi-stage filtering)
   469→                    filtered_pos = raw_pos
   470→
   471→                    # Swipe detection (direct mode)
   472→                    swipe_gesture = swipe.update(raw_pos, prev_pos, now)
   473→                    prev_pos = raw_pos.copy()
   474→
   475→                else:
   476→                    # ═══ IRONMAN TRACKING (legacy v3.1) ═══
   477→                    # Dual-stage jitter filter
   478→                    filtered_pos = np.array([
   479→                        jitter_x.filter(np.array([raw_pos[0]]))[0],
   480→                        jitter_y.filter(np.array([raw_pos[1]]))[0],
   481→                    ])
   482→
   483→                    # Velocity prediction (reduce perceived latency)
   484→                    predicted_pos = velocity_predictor.predict(filtered_pos)
   485→
   486→                    # Iron Man: finger-relative tracking
   487→                    delta = home.get_delta(predicted_pos)
   488→
   489→                    # Choose sensitivity curve based on precision mode
   490→                    active_curve = precision_curve if precision_mode else exp_curve
   491→                    mapped = active_curve.map_with_deadzone(delta, deadzone=config.deadzone)
   492→                    screen_target = np.array([
   493→                        screen_w / 2 + mapped[0] * screen_w,
   494→                        screen_h / 2 + mapped[1] * screen_h,
   495→                    ])
   496→                    screen_target[0] = np.clip(screen_target[0], 0, screen_w)
   497→                    screen_target[1] = np.clip(screen_target[1], 0, screen_h)
   498→
   499→                    # Swipe detection
   500→                    swipe_gesture = swipe.update(filtered_pos, prev_pos, now)
   501→                    prev_pos = filtered_pos.copy()
   502→
   503→                if swipe_gesture == Gesture.SWIPE_LEFT:
   504→                    _safe_kb_action(_kb(), "browser_back")
   505→                    audio.click()
   506→                elif swipe_gesture == Gesture.SWIPE_RIGHT:
   507→                    _safe_kb_action(_kb(), "browser_forward")
   508→                    audio.click()
   509→
   510→                # ═══ GESTURE ACTIONS ═══
   511→                # (works for both direct and ironman modes)
   512→
   513→                # PINCH -> Left click
   514→                if gesture == Gesture.PINCH and gesture_changed:
   515→                    if now - last_click_time > config.pinch_cooldown:
   516→                        mouse.left_click()
   517→                        audio.click()
   518→                        last_click_time = now
   519→
   520→                # PEACE -> Right click
   521→                elif gesture == Gesture.PEACE and gesture_changed:
   522→                    if now - last_click_time > config.pinch_cooldown:
   523→                        mouse.right_click()
   524→                        audio.right_click()
   525→                        last_click_time = now
   526→
   527→                # THUMBS_UP -> Double click
   528→                elif gesture == Gesture.THUMBS_UP and gesture_changed:
   529→                    mouse.double_click()
   530→                    audio.click()
   531→                    last_click_time = now
   532→
   533→                # PINKY -> Middle click
   534→                elif gesture == Gesture.PINKY and gesture_changed:
   535→                    try:
   536→                        mouse.mouse.click(mouse._button.middle, 1)
   537→                    except Exception:
   538→                        pass
   539→                    audio.right_click()
   540→                    last_click_time = now
   541→
   542→                # FIST -> Toggle freeze
   543→                elif gesture == Gesture.FIST and gesture_changed:
   544→                    cursor_frozen = not cursor_frozen
   545→                    audio.freeze()
   546→
   547→                # OK -> Close window (Alt+F4)
   548→                elif gesture == Gesture.OK and gesture_changed:
   549→                    _safe_kb_action(_kb(), "close_window")
   550→                    audio.click()
   551→
   552→                # SIX -> Task switcher (Alt+Tab)
   553→                elif gesture == Gesture.SIX and gesture_changed:
   554→                    _safe_kb_action(_kb(), "switch_window")
   555→                    audio.click()
   556→
   557→                # PALM -> Drag mode
   558→                if gesture == Gesture.PALM and not dragging:
   559→                    mouse.start_drag()
   560→                    dragging = True
   561→                    audio.drag_start()
   562→                elif gesture != Gesture.PALM and dragging:
   563→                    mouse.stop_drag()
   564→                    dragging = False
   565→
   566→                # THREE -> Scroll mode
   567→                if gesture == Gesture.THREE:
   568→                    scrolling = True
   569→                    if gesture_changed:
   570→                        prev_index_y = filtered_pos[1]
   571→                        scroll_accum = 0.0
   572→                    if prev_index_y is not None:
   573→                        sd = (filtered_pos[1] - prev_index_y) * 40
   574→                        scroll_accum += sd
   575→                        if abs(scroll_accum) > 1.0:
   576→                            mouse.scroll(int(scroll_accum))
   577→                            audio.scroll_tick()
   578→                            scroll_accum = 0.0
   579→                        prev_index_y = filtered_pos[1]
   580→                else:
   581→                    scrolling = False
   582→
   583→                # GUN -> Snap to center
   584→                if gesture == Gesture.GUN and gesture_changed:
   585→                    if is_direct:
   586→                        direct_tracker.reset(center)
   587→                    else:
   588→                        spring.reset(center)
   589→                        position_smoother.reset(center)
   590→                        home.reset()
   591→                    mouse.move_to(center[0], center[1])
   592→                    audio.click()
   593→
   594→                # ROCK -> Minimize
   595→                if gesture == Gesture.ROCK and gesture_changed:
   596→                    _safe_kb_action(_kb(), "minimize_window")
   597→                    audio.click()
   598→
   599→                # SHAKA -> Volume mode (move up/down to adjust)
   600→                #   With debounce — don't spam volume changes
   601→                if gesture == Gesture.SHAKA:
   602→                    if not volume_mode and gesture_changed:
   603→                        volume_mode = True
   604→                        audio.mode_enter()
   605→                    if gesture_changed:
   606→                        prev_index_y = filtered_pos[1]
   607→                    if prev_index_y is not None:
   608→                        vd = filtered_pos[1] - prev_index_y
   609→                        if abs(vd) > 0.02 and (now - last_volume_time > VOLUME_COOLDOWN):
   610→                            k = _kb()
   611→                            if k:
   612→                                if vd > 0:
   613→                                    _safe_kb_action(k, "volume_down")
   614→                                else:
   615→                                    _safe_kb_action(k, "volume_up")
   616→                            prev_index_y = filtered_pos[1]
   617→                            last_volume_time = now
   618→                else:
   619→                    if volume_mode:
   620→                        volume_mode = False
   621→                        audio.mode_exit()
   622→
   623→                # RING -> Brightness mode (move up/down to adjust)
   624→                #   With debounce — brightness changes are slower
   625→                if gesture == Gesture.RING:
   626→                    if not brightness_mode and gesture_changed:
   627→                        brightness_mode = True
   628→                        audio.mode_enter()
   629→                    if gesture_changed:
   630→                        prev_index_y = filtered_pos[1]
   631→                    if prev_index_y is not None:
   632→                        bd = filtered_pos[1] - prev_index_y
   633→                        if abs(bd) > 0.03 and (now - last_brightness_time > BRIGHTNESS_COOLDOWN):
   634→                            k = _kb()
   635→                            if k:
   636→                                if bd > 0:
   637→                                    _safe_kb_action(k, "brightness_down")
   638→                                else:
   639→                                    _safe_kb_action(k, "brightness_up")
   640→                            prev_index_y = filtered_pos[1]
   641→                            last_brightness_time = now
   642→                else:
   643→                    if brightness_mode:
   644→                        brightness_mode = False
   645→                        audio.mode_exit()
   646→
   647→                # ═══ CURSOR MOVEMENT ═══
   648→                if is_direct:
   649→                    # Direct mode — cursor already moved above in the direct block
   650→                    # Just handle audio feedback
   651→                    if not cursor_frozen:
   652→                        speed = np.linalg.norm(direct_tracker.velocity)
   653→                        if speed > 500:
   654→                            audio.whoosh(speed)
   655→
   656→                elif gesture in (Gesture.POINTING, Gesture.PEACE,
   657→                               Gesture.THUMBS_UP, Gesture.PINKY,
   658→                               Gesture.GUN, Gesture.ROCK) and not cursor_frozen:
   659→                    # Spring-damper physics
   660→                    cursor_pos = spring.update(screen_target, dt)
   661→                    # Edge gravity
   662→                    cursor_pos += edge_grav.apply(cursor_pos, screen_w, screen_h)
   663→                    # Momentum throw
   664→                    cursor_pos += momentum.update(spring.velocity, True, dt)
   665→                    # Clamp to screen
   666→                    cursor_pos[0] = np.clip(cursor_pos[0], 0, screen_w)
   667→                    cursor_pos[1] = np.clip(cursor_pos[1], 0, screen_h)
   668→                    # Final position smoothing for silky output
   669→                    smoothed_pos = position_smoother.smooth(cursor_pos)
   670→                    mouse.move_to(smoothed_pos[0], smoothed_pos[1])
   671→
   672→                    speed = np.linalg.norm(spring.velocity)
   673→                    if speed > 500:
   674→                        audio.whoosh(speed)
   675→
   676→                elif cursor_frozen:
   677→                    if not is_direct:
   678→                        spring.update(spring.position, dt)
   679→                        momentum.reset()
   680→
   681→                prev_gesture = gesture
   682→
   683→            else:
   684→                # No hand detected
   685→                hand_absent_frames += 1
   686→
   687→                if is_direct:
   688→                    # Direct mode — just let the spring EMA decay
   689→                    pass  # No momentum, cursor holds position
   690→                else:
   691→                    # Ironman mode — momentum throw keeps cursor gliding
   692→                    throw = momentum.update(spring.velocity, False, dt)
   693→                    if momentum.is_active:
   694→                        cp = spring.position + throw
   695→                        cp[0] = np.clip(cp[0], 0, screen_w)
   696→                        cp[1] = np.clip(cp[1], 0, screen_h)
   697→                        smoothed = position_smoother.smooth(cp)
   698→                        mouse.move_to(smoothed[0], smoothed[1])
   699→
   700→                    spring.update(spring.position, dt)
   701→
   702→                # Only reset after hand has been absent for a few frames
   703→                # This prevents jitter from momentary detection loss
   704→                if hand_absent_frames > 5:
   705→                    if not hand_was_lost:
   706→                        audio.hand_lost()
   707→                        hand_was_lost = True
   708→                    if is_direct:
   709→                        direct_tracker.reset()
   710→                    else:
   711→                        jitter_x.reset()
   712→                        jitter_y.reset()
   713→                        home.reset()
   714→                        velocity_predictor.reset()
   715→                    gsm.reset()
   716→                    prev_gesture = Gesture.NONE
   717→                    prev_index_y = None
   718→                    prev_pos = None
   719→                    if dragging:
   720→                        mouse.stop_drag()
   721→                        dragging = False
   722→                    scrolling = False
   723→                    if volume_mode:
   724→                        volume_mode = False
   725→                    if brightness_mode:
   726→                        brightness_mode = False
   727→
   728→            # Display
   729→            if config.show_camera and debug_mode and hand_data["frame"] is not None:
   730→                _draw_hud(hand_data["frame"], gesture_result, spring, fps,
   731→                          config, cursor_frozen, dragging, scrolling,
   732→                          volume_mode, brightness_mode, precision_mode,
   733→                          gsm.progress, gesture_confidence)
   734→                cv2.imshow("AirMouse", hand_data["frame"])
   735→                key = cv2.waitKey(1) & 0xFF
   736→                if key == ord("q"):
   737→                    running = False
   738→                elif key == ord("d"):
   739→                    debug_mode = not debug_mode
   740→                elif key == ord("r"):
   741→                    if is_direct:
   742→                        direct_tracker.reset(center)
   743→                    else:
   744→                        home.reset()
   745→                        spring.reset(center)
   746→                        position_smoother.reset(center)
   747→                        velocity_predictor.reset()
   748→                    audio.recalibrate()
   749→                    print("  -> Recalibrated")
   750→                elif key == ord("s"):
   751→                    config.audio_enabled = not config.audio_enabled
   752→                    audio.enabled = config.audio_enabled
   753→                    print(f"  -> Audio {'ON' if config.audio_enabled else 'OFF'}")
   754→                elif key == ord("p"):
   755→                    precision_mode = not precision_mode
   756→                    audio.precision_toggle()
   757→                    if is_direct:
   758→                        # In direct mode, precision makes it even smoother and steadier
   759→                        if precision_mode:
   760→                            direct_tracker.jitter_x = LightJitterFilter(alpha=0.20)
   761→                            direct_tracker.jitter_y = LightJitterFilter(alpha=0.20)
   762→                            direct_tracker.spring_alpha = 0.18  # Very slow, very smooth
   763→                            direct_tracker.smoother = PositionSmoother(alpha=0.50)
   764→                            direct_tracker.movement_threshold = 0.015  # Larger noise gate
   765→                            direct_tracker.pixel_deadzone = 3.0  # Larger pixel deadzone
   766→                        else:
   767→                            direct_tracker.jitter_x = LightJitterFilter(alpha=config.direct_jitter_alpha)
   768→                            direct_tracker.jitter_y = LightJitterFilter(alpha=config.direct_jitter_alpha)
   769→                            direct_tracker.spring_alpha = config.direct_spring_alpha
   770→                            direct_tracker.smoother = PositionSmoother(alpha=config.direct_smooth_alpha)
   771→                            direct_tracker.movement_threshold = config.direct_movement_threshold
   772→                            direct_tracker.pixel_deadzone = config.direct_pixel_deadzone
   773→                    else:
   774→                        if precision_mode:
   775→                            position_smoother = PositionSmoother(alpha=0.6)  # Smoother in precision mode
   776→                        else:
   777→                            position_smoother = PositionSmoother(alpha=config.position_smooth_alpha)
   778→                    print(f"  -> Precision mode {'ON' if precision_mode else 'OFF'}"
   779→                          f" (power={'1.0' if precision_mode else config.exp_power},"
   780→                          f" scale={'1.0' if precision_mode else config.exp_scale})")
   781→                elif key == ord("t"):
   782→                    run_tutorial(tracker)
   783→                elif key == ord("h"):
   784→                    _show_quick_reference()
   785→
   786→            # FPS throttling
   787→            elapsed = time.perf_counter() - t0
   788→            frame_times.append(elapsed)
   789→            if len(frame_times) > 30:
   790→                frame_times.pop(0)
   791→            avg = sum(frame_times) / len(frame_times)
   792→            fps = 1.0 / max(avg, 1e-6)
   793→            min_frame = 1.0 / config.target_fps
   794→            if elapsed < min_frame:
   795→                time.sleep(min_frame - elapsed)
   796→
   797→    except KeyboardInterrupt:
   798→        print("\n  Shutting down...")
   799→    finally:
   800→        running = False
   801→        tracker.release()
   802→        if dragging:
   803→            try:
   804→                mouse.stop_drag()
   805→            except Exception:
   806→                pass
   807→        try:
   808→            if config.show_camera:
   809→                cv2.destroyAllWindows()
   810→        except Exception:
   811→            pass
   812→        print("  AirMouse stopped.")
   813→
   814→
   815→if __name__ == "__main__":
   816→    main()
