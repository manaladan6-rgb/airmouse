    47→def _get_screen_size():
    48→    try:
    49→        import ctypes
    50→        user32 = ctypes.windll.user32
    51→        w, h = user32.GetSystemMetrics(0), user32.GetSystemMetrics(1)
    52→        if w > 0 and h > 0:
    53→            return w, h
    54→    except Exception:
    55→        pass
    56→    try:
    57→        import tkinter as tk
    58→        root = tk.Tk()
    59→        w, h = root.winfo_screenwidth(), root.winfo_screenheight()
    60→        root.destroy()
    61→        if w > 0 and h > 0:
    62→            return w, h
    63→    except Exception:
    64→        pass
    65→    return 1920, 1080

    68→def _draw_hud(frame, gesture_result, spring, fps, config,
    69→              frozen, dragging, scrolling, volume_mode, brightness_mode,
    70→              precision_mode, gsm_progress, gesture_confidence):
    71→    """Draw Iron Man HUD overlay with v3.1 enhancements."""
    72→    if frame is None:
    73→        return
    74→    h, w = frame.shape[:2]
    75→    gesture = gesture_result.get("gesture", "none")
    76→
    77→    colors = {
    78→        Gesture.POINTING: (0, 255, 255), Gesture.PINCH: (0, 255, 0),
    79→        Gesture.PEACE: (0, 255, 200), Gesture.PALM: (255, 128, 0),
    80→        Gesture.FIST: (0, 0, 255), Gesture.THUMBS_UP: (0, 200, 0),
    81→        Gesture.THREE: (255, 0, 255), Gesture.PINKY: (200, 200, 0),
    82→        Gesture.GUN: (0, 200, 200), Gesture.ROCK: (200, 0, 200),
    83→        Gesture.SHAKA: (0, 200, 100), Gesture.OK: (0, 215, 255),
    84→        Gesture.RING: (255, 200, 0), Gesture.SIX: (100, 200, 200),
    85→        "none": (80, 80, 80),
    86→    }
    87→    color = colors.get(gesture, (80, 80, 80))
    88→
    89→    # Gesture indicator (top-left) with confirm progress ring
    90→    cx, cy = 50, 50
    91→    # Outer ring — progress indicator
    92→    if gsm_progress > 0 and gsm_progress < 1.0:
    93→        # Draw progress arc
    94→        end_angle = int(360 * gsm_progress)
    95→        cv2.ellipse(frame, (cx, cy), (30, 30), -90, 0, end_angle, (0, 255, 255), 3)
    96→        cv2.circle(frame, (cx, cy), 30, (40, 40, 40), 1)
    97→    else:
    98→        cv2.circle(frame, (cx, cy), 28, color, 3)
    99→    cv2.circle(frame, (cx, cy), 4, color, -1)
   100→
   101→    info = GESTURE_INFO.get(gesture)
   102→    label = info["name"] if info else gesture
   103→    cv2.putText(frame, label.upper(), (cx + 38, cy + 8),
   104→                cv2.FONT_HERSHEY_SIMPLEX, 0.65, color, 2, cv2.LINE_AA)
   105→
   106→    # Confidence bar (next to gesture name)
   107→    if gesture_confidence > 0:
   108→        bar_x = cx + 38 + len(label) * 12 + 10
   109→        bar_w = 40
   110→        cv2.rectangle(frame, (bar_x, cy - 6), (bar_x + bar_w, cy + 6), (40, 40, 40), -1)
   111→        fill_w = int(bar_w * gesture_confidence)
   112→        conf_color = (0, 255, 0) if gesture_confidence > 0.7 else (0, 200, 255)
   113→        cv2.rectangle(frame, (bar_x, cy - 6), (bar_x + fill_w, cy + 6), conf_color, -1)
   114→
   115→    # Status badges (bottom)
   116→    badges = []
   117→    if frozen:
   118→        badges.append(("FROZEN", (0, 0, 255)))
   119→    if dragging:
   120→        badges.append(("DRAG", (255, 128, 0)))
   121→    if scrolling:
   122→        badges.append(("SCROLL", (255, 0, 255)))
   123→    if volume_mode:
   124→        badges.append(("VOL", (0, 200, 100)))
   125→    if brightness_mode:
   126→        badges.append(("BRIGHT", (255, 200, 0)))
   127→    if precision_mode:
   128→        badges.append(("PRECISE", (200, 200, 255)))
   129→    bx = 10
   130→    for bt, bc in badges:
   131→        # Badge background
   132→        (tw, _), _ = cv2.getTextSize(bt, cv2.FONT_HERSHEY_SIMPLEX, 0.55, 2)
   133→        cv2.rectangle(frame, (bx - 3, h - 38), (bx + tw + 5, h - 12), (20, 20, 20), -1)
   134→        cv2.putText(frame, bt, (bx, h - 20),
   135→                    cv2.FONT_HERSHEY_SIMPLEX, 0.55, bc, 2, cv2.LINE_AA)
   136→        bx += tw + 18
   137→
   138→    # Physics stats (bottom-right)
   139→    k = spring.current_stiffness
   140→    speed = np.linalg.norm(spring.velocity)
   141→    stats = [f"FPS:{fps:.0f}", f"k:{k:.0f}", f"v:{speed:.0f}"]
   142→    for i, t in enumerate(stats):
   143→        cv2.putText(frame, t, (w - 120, h - 55 + i * 20),
   144→                    cv2.FONT_HERSHEY_SIMPLEX, 0.45, (0, 255, 0), 1, cv2.LINE_AA)
   145→
   146→    # Draw tracked finger landmarks
   147→    lm = gesture_result.get("landmarks")
   148→    if lm is not None:
   149→        from .gestures import INDEX_TIP, THUMB_TIP
   150→        tip = lm[INDEX_TIP]
   151→        tx, ty = int(tip.x * w), int(tip.y * h)
   152→
   153→        # Cursor crosshair with glow
   154→        glow_color = (0, 80, 80) if not precision_mode else (80, 80, 0)
   155→        cv2.circle(frame, (tx, ty), 11, glow_color, -1)  # Glow
   156→        cv2.circle(frame, (tx, ty), 7, (0, 255, 255) if not precision_mode else (255, 255, 0), -1)
   157→        cv2.circle(frame, (tx, ty), 13, (0, 255, 255) if not precision_mode else (255, 255, 0), 2)
   158→
   159→        thumb = lm[THUMB_TIP]
   160→        thx, thy = int(thumb.x * w), int(thumb.y * h)
   161→        cv2.circle(frame, (thx, thy), 4, (255, 200, 0), -1)
   162→        if gesture == Gesture.PINCH:
   163→            cv2.line(frame, (tx, ty), (thx, thy), (0, 255, 0), 2)

   184→def _safe_kb_action(kb_instance, action_name):
   185→    """Safely call a keyboard action — never crash the main loop."""
   186→    if kb_instance is None:
   187→        return
   188→    try:
   189→        method = getattr(kb_instance, action_name)
   190→        method()
   191→    except Exception:
   192→        pass

   209→    parser.add_argument("--list-monitors", action="store_true", help="List monitors and exit")
   210→    parser.add_argument("--autostart", type=str, choices=["on", "off"], default=None, help="Enable/disable auto-start")
   211→    parser.add_argument("--settings", action="store_true", help="Open settings GUI")
   212→    args = parser.parse_args()
   213→
   214→    # Handle --settings
   215→    if args.settings:
   216→        from .settings_gui import show_settings
   217→        show_settings()

   219→
   220→    # Handle --list-monitors
   221→    if args.list_monitors:
   222→        displays = enumerate_displays()
   223→        print("  Connected displays:")
   224→        for d in displays:
   225→            print(f"    [{d.index}] {d.width}x{d.height} at ({d.x},{d.y}) {'[PRIMARY]' if d.is_primary else ''} — {d.name}")
   226→        return
   227→
   228→    # Handle --autostart
   229→    if args.autostart is not None:
   230→        from .autostart import enable_auto_start, disable_auto_start, is_auto_start_enabled
   231→        if args.autostart == "on":
   232→            if enable_auto_start():
   233→                print("  Auto-start ENABLED — AirMouse will start on boot.")
   234→            else:
   235→                print("  Failed to enable auto-start.")
   236→        else:
   237→            if disable_auto_start():
   238→                print("  Auto-start DISABLED.")
   239→            else:
   240→                print("  Failed to disable auto-start.")
   241→        return
   242→
   243→    config = Config()
   244→    config.load()
   245→    if not os.path.exists(CONFIG_PATH):
   246→        config.save_defaults()
   247→
   248→    if args.no_cam: config.show_camera = False
   249→    if args.no_sound: config.audio_enabled = False
   250→    if args.cam is not None: config.camera_index = args.cam
   251→    if args.power is not None: config.exp_power = args.power
   252→    if args.scale is not None: config.exp_scale = args.scale
   253→
   254→    # Multi-monitor support
   255→    displays = enumerate_displays()
   256→    selected_display = None
   257→    if args.monitor is not None and args.monitor < len(displays):
   258→        selected_display = displays[args.monitor]
   259→    else:
   260→        selected_display = get_primary_display()
   261→    screen_w = selected_display.width
   262→    screen_h = selected_display.height
   263→
   264→    tracker = HandTracker(
   265→        camera_index=config.camera_index,
   266→        detection_confidence=config.detection_confidence,
   267→        tracking_confidence=config.tracking_confidence,
   268→    )
   269→
   270→    mouse = MouseController(screen_w=screen_w, screen_h=screen_h)
   271→    audio = AudioFeedback(enabled=config.audio_enabled)
   272→
   273→    # Tutorial check
   274→    tutorial_done_file = os.path.join(os.path.expanduser("~"), ".airmouse", "tutorial_done")
   275→    should_tutorial = args.tutorial or (not args.skip and not os.path.exists(tutorial_done_file))
   276→
   277→    if should_tutorial:
   278→        completed = run_tutorial(tracker)
   279→        if completed:
   280→            os.makedirs(os.path.dirname(tutorial_done_file), exist_ok=True)
   281→            with open(tutorial_done_file, "w") as f:
   282→                f.write("done")
   283→
   284→    # ═══ Physics stack v3.1 ═══
   285→    spring = AdaptiveSpringDamper(
   286→        mass=config.mass, stiffness_min=config.stiffness_min,
   287→        stiffness_max=config.stiffness_max, damping_ratio=config.damping_ratio,
   288→        speed_threshold=config.speed_threshold,
   289→        max_accel=config.max_accel,
   290→        stiffness_smoothing=config.stiffness_smoothing,
   291→    )
   292→    # Dual-stage jitter filter (micro-tremor + macro smooth)
   293→    jitter_x = DualStageJitterFilter(
   294→        micro_alpha=config.jitter_micro_alpha,
   295→        macro_alpha=config.jitter_macro_alpha,
   296→    )
   297→    jitter_y = DualStageJitterFilter(
   298→        micro_alpha=config.jitter_micro_alpha,
   299→        macro_alpha=config.jitter_macro_alpha,
   300→    )
   301→    home = HomePosition(
   302→        drift_rate=config.home_drift_rate,
   303→        drift_rate_moving=config.home_drift_rate_moving,
   304→    )
   305→    exp_curve = ExponentialCurve(power=config.exp_power, scale=config.exp_scale)
   306→    precision_curve = ExponentialCurve(power=config.precision_power, scale=config.precision_scale)
   307→    momentum = MomentumThrow(
   308→        friction=config.throw_friction,
   309→        min_speed=config.throw_min_speed,
   310→        max_momentum=config.throw_max_momentum,
   311→    )
   312→    edge_grav = EdgeGravity(
   313→        strength=config.edge_gravity_strength,
   314→        edge_zone=config.edge_gravity_zone,
   315→    )
   316→    # v3.1 new physics components
   317→    velocity_predictor = VelocityPredictor(
   318→        prediction_factor=config.prediction_factor,
   319→        max_correction=config.prediction_max_correction,
   320→    )
   321→    position_smoother = PositionSmoother(alpha=config.position_smooth_alpha)
   322→
   323→    swipe = SwipeDetector()
   324→    gsm = GestureStateMachine(
   325→        confirm_frames=config.gesture_confirm_frames,
   326→        action_confirm_frames=config.gesture_action_confirm_frames,
   327→        transition_cooldown=config.gesture_transition_cooldown,
   328→    )
   329→
   330→    # State
   331→    center = np.array([screen_w / 2.0, screen_h / 2.0])
   332→    spring.reset(center)
   333→    position_smoother.reset(center)
   334→    mouse.move_to(center[0], center[1])
   335→
   336→    last_click_time = 0.0
   337→    prev_gesture = Gesture.NONE
   338→    cursor_frozen = False
   339→    dragging = False
   340→    scrolling = False
   341→    volume_mode = False
   342→    brightness_mode = False
   343→    precision_mode = args.precision or config.precision_mode
   344→    scroll_accum = 0.0
   345→    prev_index_y = None
   346→    prev_pos = None
   347→    fps = 0.0
   348→    frame_times = []
   349→    debug_mode = True
   350→    running = True
   351→    gesture_confidence = 0.0
   352→    hand_absent_frames = 0  # Track how long hand has been absent
   353→    hand_was_lost = False    # Track hand-loss for sound feedback
   354→
   355→    # Volume/brightness mode debounce
   356→    last_volume_time = 0.0
   357→    last_brightness_time = 0.0
   358→    VOLUME_COOLDOWN = 0.15   # Seconds between volume adjustments
   359→    BRIGHTNESS_COOLDOWN = 0.3  # Brightness changes are slower
   360→
   361→    # Keyboard actions (lazy-init, never crash)
   362→    kb = None
   363→    def _kb():
   364→        nonlocal kb
   365→        if kb is None:
   366→            try:
   367→                from .keyboard import KeyboardActions
   368→                kb = KeyboardActions()
   369→            except Exception:
   370→                pass
   371→        return kb
   372→
   373→    # Banner
   374→    print()
   375→    print("  +==================================================+")
   376→    print(f"  |     AirMouse v{_pkg.__version__} - Iron Man Next-Gen Edition  |")
   377→    print("  |     14 gestures | Physics cursor | Finger-relative  |")
   378→    print("  +==================================================+")
   379→    print(f"  Screen: {screen_w}x{screen_h}  |  Audio: {'ON' if config.audio_enabled else 'OFF'}  |  Gestures: 14+2 swipe")
   380→    if precision_mode:
   381→        print(f"  Mode: PRECISION (power={config.precision_power}, scale={config.precision_scale})")
   382→    _show_quick_reference()
   383→    print()
   384→
   385→    try:
   386→        while running:
   387→            t0 = time.perf_counter()
   388→            dt = 1.0 / max(fps, 1.0)
   389→            now = time.perf_counter()
   390→
   391→            hand_data = tracker.read()
   392→            gesture_result = {"gesture": Gesture.NONE, "landmarks": None, "confidence": 0.0}
   393→
   394→            if hand_data["hand_found"] and hand_data["landmarks"] is not None:
   395→                hand_absent_frames = 0  # Reset absent counter
   396→                if hand_was_lost:
   397→                    audio.hand_found()
   398→                    hand_was_lost = False
   399→                gesture_result = recognize_gesture(hand_data["landmarks"],
   400→                                                    pinch_threshold=config.pinch_threshold)
   401→                raw_gesture = gesture_result["gesture"]
   402→                gesture_confidence = gesture_result.get("confidence", 0.0)
   403→
   404→                # Gesture state machine — with stability and time
   405→                hand_stable = home.is_stable() if home.is_calibrated else False
   406→                gesture = gsm.update(raw_gesture, now=now, hand_stable=hand_stable)
   407→                gesture_changed = (gesture != prev_gesture)
   408→
   409→                # Sound feedback on gesture confirmation
   410→                if gesture_changed and gesture != Gesture.NONE:
   411→                    if prev_gesture == Gesture.NONE or gsm.progress >= 1.0:
   412→                        audio.gesture_confirm()
   413→
   414→                raw_pos = gesture_result["index_pos"]
   415→
   416→                # Dual-stage jitter filter
   417→                filtered_pos = np.array([
   418→                    jitter_x.filter(np.array([raw_pos[0]]))[0],
   419→                    jitter_y.filter(np.array([raw_pos[1]]))[0],
   420→                ])
   421→
   422→                # Velocity prediction (reduce perceived latency)
   423→                predicted_pos = velocity_predictor.predict(filtered_pos)
   424→
   425→                # Iron Man: finger-relative tracking
   426→                delta = home.get_delta(predicted_pos)
   427→
   428→                # Choose sensitivity curve based on precision mode
   429→                active_curve = precision_curve if precision_mode else exp_curve
   430→                mapped = active_curve.map_with_deadzone(delta, deadzone=config.deadzone)
   431→                screen_target = np.array([
   432→                    screen_w / 2 + mapped[0] * screen_w,
   433→                    screen_h / 2 + mapped[1] * screen_h,
   434→                ])
   435→                screen_target[0] = np.clip(screen_target[0], 0, screen_w)
   436→                screen_target[1] = np.clip(screen_target[1], 0, screen_h)
   437→
   438→                # Swipe detection
   439→                swipe_gesture = swipe.update(filtered_pos, prev_pos, now)
   440→                prev_pos = filtered_pos.copy()
   441→
   442→                if swipe_gesture == Gesture.SWIPE_LEFT:
   443→                    _safe_kb_action(_kb(), "browser_back")
   444→                    audio.click()
   445→                elif swipe_gesture == Gesture.SWIPE_RIGHT:
   446→                    _safe_kb_action(_kb(), "browser_forward")
   447→                    audio.click()
   448→
   449→                # ═══ GESTURE ACTIONS ═══
   450→
   451→                # PINCH -> Left click
   452→                if gesture == Gesture.PINCH and gesture_changed:
   453→                    if now - last_click_time > config.pinch_cooldown:
   454→                        mouse.left_click()
   455→                        audio.click()
   456→                        last_click_time = now
   457→
   458→                # PEACE -> Right click
   459→                elif gesture == Gesture.PEACE and gesture_changed:
   460→                    if now - last_click_time > config.pinch_cooldown:
   461→                        mouse.right_click()
   462→                        audio.right_click()
   463→                        last_click_time = now
   464→
   465→                # THUMBS_UP -> Double click
   466→                elif gesture == Gesture.THUMBS_UP and gesture_changed:
   467→                    mouse.double_click()
   468→                    audio.click()
   469→                    last_click_time = now
   470→
   471→                # PINKY -> Middle click
   472→                elif gesture == Gesture.PINKY and gesture_changed:
   473→                    try:
   474→                        mouse.mouse.click(mouse._button.middle, 1)
   475→                    except Exception:
   476→                        pass
   477→                    audio.right_click()
   478→                    last_click_time = now
   479→
   480→                # FIST -> Toggle freeze
   481→                elif gesture == Gesture.FIST and gesture_changed:
   482→                    cursor_frozen = not cursor_frozen
   483→                    audio.freeze()
   484→
   485→                # OK -> Close window (Alt+F4)
   486→                elif gesture == Gesture.OK and gesture_changed:
   487→                    _safe_kb_action(_kb(), "close_window")
   488→                    audio.click()
   489→
   490→                # SIX -> Task switcher (Alt+Tab)
   491→                elif gesture == Gesture.SIX and gesture_changed:
   492→                    _safe_kb_action(_kb(), "switch_window")
   493→                    audio.click()
   494→
   495→                # PALM -> Drag mode
   496→                if gesture == Gesture.PALM and not dragging:
   497→                    mouse.start_drag()
   498→                    dragging = True
   499→                    audio.drag_start()
   500→                elif gesture != Gesture.PALM and dragging:
   501→                    mouse.stop_drag()
   502→                    dragging = False
   503→
   504→                # THREE -> Scroll mode
   505→                if gesture == Gesture.THREE:
   506→                    scrolling = True
   507→                    if gesture_changed:
   508→                        prev_index_y = filtered_pos[1]
   509→                        scroll_accum = 0.0
   510→                    if prev_index_y is not None:
   511→                        sd = (filtered_pos[1] - prev_index_y) * 40
   512→                        scroll_accum += sd
   513→                        if abs(scroll_accum) > 1.0:
   514→                            mouse.scroll(int(scroll_accum))
   515→                            audio.scroll_tick()
   516→                            scroll_accum = 0.0
   517→                        prev_index_y = filtered_pos[1]
   518→                else:
   519→                    scrolling = False
   520→
   521→                # GUN -> Snap to center
   522→                if gesture == Gesture.GUN and gesture_changed:
   523→                    spring.reset(center)
   524→                    position_smoother.reset(center)
   525→                    mouse.move_to(center[0], center[1])
   526→                    home.reset()
   527→                    audio.click()
   528→
   529→                # ROCK -> Minimize
   530→                if gesture == Gesture.ROCK and gesture_changed:
   531→                    _safe_kb_action(_kb(), "minimize_window")
   532→                    audio.click()
   533→
   534→                # SHAKA -> Volume mode (move up/down to adjust)
   535→                #   With debounce — don't spam volume changes
   536→                if gesture == Gesture.SHAKA:
   537→                    if not volume_mode and gesture_changed:
   538→                        volume_mode = True
   539→                        audio.mode_enter()
   540→                    if gesture_changed:
   541→                        prev_index_y = filtered_pos[1]
   542→                    if prev_index_y is not None:
   543→                        vd = filtered_pos[1] - prev_index_y
   544→                        if abs(vd) > 0.02 and (now - last_volume_time > VOLUME_COOLDOWN):
   545→                            k = _kb()
   546→                            if k:
   547→                                if vd > 0:
   548→                                    _safe_kb_action(k, "volume_down")
   549→                                else:
   550→                                    _safe_kb_action(k, "volume_up")
   551→                            prev_index_y = filtered_pos[1]
   552→                            last_volume_time = now
   553→                else:
   554→                    if volume_mode:
   555→                        volume_mode = False
   556→                        audio.mode_exit()
   557→
   558→                # RING -> Brightness mode (move up/down to adjust)
   559→                #   With debounce — brightness changes are slower
   560→                if gesture == Gesture.RING:
   561→                    if not brightness_mode and gesture_changed:
   562→                        brightness_mode = True
   563→                        audio.mode_enter()
   564→                    if gesture_changed:
   565→                        prev_index_y = filtered_pos[1]
   566→                    if prev_index_y is not None:
   567→                        bd = filtered_pos[1] - prev_index_y
   568→                        if abs(bd) > 0.03 and (now - last_brightness_time > BRIGHTNESS_COOLDOWN):
   569→                            k = _kb()
   570→                            if k:
   571→                                if bd > 0:
   572→                                    _safe_kb_action(k, "brightness_down")
   573→                                else:
   574→                                    _safe_kb_action(k, "brightness_up")
   575→                            prev_index_y = filtered_pos[1]
   576→                            last_brightness_time = now
   577→                else:
   578→                    if brightness_mode:
   579→                        brightness_mode = False
   580→                        audio.mode_exit()
   581→
   582→                # ═══ CURSOR MOVEMENT ═══
   583→                # Pointing and navigation gestures move cursor
   584→                if gesture in (Gesture.POINTING, Gesture.PEACE,
   585→                               Gesture.THUMBS_UP, Gesture.PINKY,
   586→                               Gesture.GUN, Gesture.ROCK) and not cursor_frozen:
   587→                    # Spring-damper physics
   588→                    cursor_pos = spring.update(screen_target, dt)
   589→                    # Edge gravity
   590→                    cursor_pos += edge_grav.apply(cursor_pos, screen_w, screen_h)
   591→                    # Momentum throw
   592→                    cursor_pos += momentum.update(spring.velocity, True, dt)
   593→                    # Clamp to screen
   594→                    cursor_pos[0] = np.clip(cursor_pos[0], 0, screen_w)
   595→                    cursor_pos[1] = np.clip(cursor_pos[1], 0, screen_h)
   596→                    # Final position smoothing for silky output
   597→                    smoothed_pos = position_smoother.smooth(cursor_pos)
   598→                    mouse.move_to(smoothed_pos[0], smoothed_pos[1])
   599→
   600→                    speed = np.linalg.norm(spring.velocity)
   601→                    if speed > 500:
   602→                        audio.whoosh(speed)
   603→
   604→                elif cursor_frozen:
   605→                    spring.update(spring.position, dt)
   606→                    momentum.reset()
   607→
   608→                prev_gesture = gesture
   609→
   610→            else:
   611→                # No hand detected
   612→                hand_absent_frames += 1
   613→
   614→                # Momentum throw keeps cursor gliding
   615→                throw = momentum.update(spring.velocity, False, dt)
   616→                if momentum.is_active:
   617→                    cp = spring.position + throw
   618→                    cp[0] = np.clip(cp[0], 0, screen_w)
   619→                    cp[1] = np.clip(cp[1], 0, screen_h)
   620→                    smoothed = position_smoother.smooth(cp)
   621→                    mouse.move_to(smoothed[0], smoothed[1])
   622→
   623→                spring.update(spring.position, dt)
   624→
   625→                # Only reset after hand has been absent for a few frames
   626→                # This prevents jitter from momentary detection loss
   627→                if hand_absent_frames > 5:
   628→                    if not hand_was_lost:
   629→                        audio.hand_lost()
   630→                        hand_was_lost = True
   631→                    jitter_x.reset()
   632→                    jitter_y.reset()
   633→                    home.reset()
   634→                    velocity_predictor.reset()
   635→                    gsm.reset()
   636→                    prev_gesture = Gesture.NONE
   637→                    prev_index_y = None
   638→                    prev_pos = None
   639→                    if dragging:
   640→                        mouse.stop_drag()
   641→                        dragging = False
   642→                    scrolling = False
   643→                    if volume_mode:
   644→                        volume_mode = False
   645→                    if brightness_mode:
   646→                        brightness_mode = False
   647→
   648→            # Display
   649→            if config.show_camera and debug_mode and hand_data["frame"] is not None:
   650→                _draw_hud(hand_data["frame"], gesture_result, spring, fps,
   651→                          config, cursor_frozen, dragging, scrolling,
   652→                          volume_mode, brightness_mode, precision_mode,
   653→                          gsm.progress, gesture_confidence)
   654→                cv2.imshow("AirMouse", hand_data["frame"])
   655→                key = cv2.waitKey(1) & 0xFF
   656→                if key == ord("q"):
   657→                    running = False
   658→                elif key == ord("d"):
   659→                    debug_mode = not debug_mode
   660→                elif key == ord("r"):
   661→                    home.reset()
   662→                    spring.reset(center)
   663→                    position_smoother.reset(center)
   664→                    velocity_predictor.reset()
   665→                    audio.recalibrate()
   666→                    print("  -> Recalibrated")
   667→                elif key == ord("s"):
   668→                    config.audio_enabled = not config.audio_enabled
   669→                    audio.enabled = config.audio_enabled
   670→                    print(f"  -> Audio {'ON' if config.audio_enabled else 'OFF'}")
   671→                elif key == ord("p"):
   672→                    precision_mode = not precision_mode
   673→                    audio.precision_toggle()
   674→                    if precision_mode:
   675→                        position_smoother = PositionSmoother(alpha=0.6)  # Smoother in precision mode
   676→                    else:
   677→                        position_smoother = PositionSmoother(alpha=config.position_smooth_alpha)
   678→                    print(f"  -> Precision mode {'ON' if precision_mode else 'OFF'}"
   679→                          f" (power={'1.0' if precision_mode else config.exp_power},"
   680→                          f" scale={'1.0' if precision_mode else config.exp_scale})")
   681→                elif key == ord("t"):
   682→                    run_tutorial(tracker)
   683→                elif key == ord("h"):
   684→                    _show_quick_reference()
   685→
   686→            # FPS throttling
   687→            elapsed = time.perf_counter() - t0
   688→            frame_times.append(elapsed)
   689→            if len(frame_times) > 30:
   690→                frame_times.pop(0)
   691→            avg = sum(frame_times) / len(frame_times)
   692→            fps = 1.0 / max(avg, 1e-6)
   693→            min_frame = 1.0 / config.target_fps
   694→            if elapsed < min_frame:
   695→                time.sleep(min_frame - elapsed)
   696→
   697→    except KeyboardInterrupt:
   698→        print("\n  Shutting down...")
   699→    finally:
   700→        running = False
   701→        tracker.release()
   702→        if dragging:
   703→            try:
   704→                mouse.stop_drag()
   705→            except Exception:
   706→                pass
   707→        try:
   708→            if config.show_camera:
   709→                cv2.destroyAllWindows()
   710→        except Exception:
   711→            pass
   712→        print("  AirMouse stopped.")
   713→
   714→
   715→if __name__ == "__main__":
   716→    main()
