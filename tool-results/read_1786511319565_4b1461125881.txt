    68→    68→def _draw_hud(frame, gesture_result, spring, fps, config,
    69→    69→              frozen, dragging, scrolling, volume_mode, brightness_mode,
    70→    70→              precision_mode, gsm_progress, gesture_confidence):
    71→    71→    """Draw Iron Man HUD overlay with v3.1 enhancements."""
    72→    72→    if frame is None:
    73→    73→        return
    74→    74→    h, w = frame.shape[:2]
    75→    75→    gesture = gesture_result.get("gesture", "none")
    76→    76→
    77→    77→    colors = {
    78→    78→        Gesture.POINTING: (0, 255, 255), Gesture.PINCH: (0, 255, 0),
    79→    79→        Gesture.PEACE: (0, 255, 200), Gesture.PALM: (255, 128, 0),
    80→    80→        Gesture.FIST: (0, 0, 255), Gesture.THUMBS_UP: (0, 200, 0),
    81→    81→        Gesture.THREE: (255, 0, 255), Gesture.PINKY: (200, 200, 0),
    82→    82→        Gesture.GUN: (0, 200, 200), Gesture.ROCK: (200, 0, 200),
    83→    83→        Gesture.SHAKA: (0, 200, 100), Gesture.OK: (0, 215, 255),
    84→    84→        Gesture.RING: (255, 200, 0), Gesture.SIX: (100, 200, 200),
    85→    85→        "none": (80, 80, 80),
    86→    86→    }
    87→    87→    color = colors.get(gesture, (80, 80, 80))
    88→    88→
    89→    89→    # Gesture indicator (top-left) with confirm progress ring
    90→    90→    cx, cy = 50, 50
    91→    91→    # Outer ring — progress indicator
    92→    92→    if gsm_progress > 0 and gsm_progress < 1.0:
    93→    93→        # Draw progress arc
    94→    94→        end_angle = int(360 * gsm_progress)
    95→    95→        cv2.ellipse(frame, (cx, cy), (30, 30), -90, 0, end_angle, (0, 255, 255), 3)
    96→    96→        cv2.circle(frame, (cx, cy), 30, (40, 40, 40), 1)
    97→    97→    else:
    98→    98→        cv2.circle(frame, (cx, cy), 28, color, 3)
    99→    99→    cv2.circle(frame, (cx, cy), 4, color, -1)
   100→   100→
   101→   101→    info = GESTURE_INFO.get(gesture)
   102→   102→    label = info["name"] if info else gesture
   103→   103→    cv2.putText(frame, label.upper(), (cx + 38, cy + 8),
   104→   104→                cv2.FONT_HERSHEY_SIMPLEX, 0.65, color, 2, cv2.LINE_AA)
   105→   105→
   106→   106→    # Confidence bar (next to gesture name)
   107→   107→    if gesture_confidence > 0:
   108→   108→        bar_x = cx + 38 + len(label) * 12 + 10
   109→   109→        bar_w = 40
   110→   110→        cv2.rectangle(frame, (bar_x, cy - 6), (bar_x + bar_w, cy + 6), (40, 40, 40), -1)
   111→   111→        fill_w = int(bar_w * gesture_confidence)
   112→   112→        conf_color = (0, 255, 0) if gesture_confidence > 0.7 else (0, 200, 255)
   113→   113→        cv2.rectangle(frame, (bar_x, cy - 6), (bar_x + fill_w, cy + 6), conf_color, -1)
   114→   114→
   115→   115→    # Status badges (bottom)
   116→   116→    badges = []
   117→   117→    if frozen:
   118→   118→        badges.append(("FROZEN", (0, 0, 255)))
   119→   119→    if dragging:
   120→   120→        badges.append(("DRAG", (255, 128, 0)))
   121→   121→    if scrolling:
   122→   122→        badges.append(("SCROLL", (255, 0, 255)))
   123→   123→    if volume_mode:
   124→   124→        badges.append(("VOL", (0, 200, 100)))
   125→   125→    if brightness_mode:
   126→   126→        badges.append(("BRIGHT", (255, 200, 0)))
   127→   127→    if precision_mode:
   128→   128→        badges.append(("PRECISE", (200, 200, 255)))
   129→   129→    bx = 10
   130→   130→    for bt, bc in badges:
   131→   131→        # Badge background
   132→   132→        (tw, _), _ = cv2.getTextSize(bt, cv2.FONT_HERSHEY_SIMPLEX, 0.55, 2)
   133→   133→        cv2.rectangle(frame, (bx - 3, h - 38), (bx + tw + 5, h - 12), (20, 20, 20), -1)
   134→   134→        cv2.putText(frame, bt, (bx, h - 20),
   135→   135→                    cv2.FONT_HERSHEY_SIMPLEX, 0.55, bc, 2, cv2.LINE_AA)
   136→   136→        bx += tw + 18
   137→   137→
   138→   138→    # Physics stats (bottom-right)
   139→   139→    k = spring.current_stiffness
   140→   140→    speed = np.linalg.norm(spring.velocity)
   141→   141→    stats = [f"FPS:{fps:.0f}", f"k:{k:.0f}", f"v:{speed:.0f}"]
   142→   142→    for i, t in enumerate(stats):
   143→   143→        cv2.putText(frame, t, (w - 120, h - 55 + i * 20),
   144→   144→                    cv2.FONT_HERSHEY_SIMPLEX, 0.45, (0, 255, 0), 1, cv2.LINE_AA)
   145→   145→
   146→   146→    # Draw tracked finger landmarks
   147→   147→    lm = gesture_result.get("landmarks")
   148→   148→    if lm is not None:
   149→   149→        from .gestures import INDEX_TIP, THUMB_TIP
   150→   150→        tip = lm[INDEX_TIP]
   151→   151→        tx, ty = int(tip.x * w), int(tip.y * h)
   152→   152→
   153→   153→        # Cursor crosshair with glow
   154→   154→        glow_color = (0, 80, 80) if not precision_mode else (80, 80, 0)
   155→   155→        cv2.circle(frame, (tx, ty), 11, glow_color, -1)  # Glow
   156→   156→        cv2.circle(frame, (tx, ty), 7, (0, 255, 255) if not precision_mode else (255, 255, 0), -1)
   157→   157→        cv2.circle(frame, (tx, ty), 13, (0, 255, 255) if not precision_mode else (255, 255, 0), 2)
   158→   158→
   159→   159→        thumb = lm[THUMB_TIP]
   160→   160→        thx, thy = int(thumb.x * w), int(thumb.y * h)
   161→   161→        cv2.circle(frame, (thx, thy), 4, (255, 200, 0), -1)
   162→   162→        if gesture == Gesture.PINCH:
   163→   163→            cv2.line(frame, (tx, ty), (thx, thy), (0, 255, 0), 2)

   184→   184→def _safe_kb_action(kb_instance, action_name):
   185→   185→    """Safely call a keyboard action — never crash the main loop."""
   186→   186→    if kb_instance is None:
   187→   187→        return
   188→   188→    try:
   189→   189→        method = getattr(kb_instance, action_name)
   190→   190→        method()
   191→   191→    except Exception:
   192→   192→        pass

   419→   419→                    jitter_y.filter(np.array([raw_pos[1]]))[0],
   420→   420→                ])
   421→   421→
   422→   422→                # Velocity prediction (reduce perceived latency)
   423→   423→                predicted_pos = velocity_predictor.predict(filtered_pos)
   424→   424→
   425→   425→                # Iron Man: finger-relative tracking
   426→   426→                delta = home.get_delta(predicted_pos)
   427→   427→
   428→   428→                # Choose sensitivity curve based on precision mode
   429→   429→                active_curve = precision_curve if precision_mode else exp_curve
   430→   430→                mapped = active_curve.map_with_deadzone(delta, deadzone=config.deadzone)
   431→   431→                screen_target = np.array([
   432→   432→                    screen_w / 2 + mapped[0] * screen_w,
   433→   433→                    screen_h / 2 + mapped[1] * screen_h,
   434→   434→                ])
   435→   435→                screen_target[0] = np.clip(screen_target[0], 0, screen_w)
   436→   436→                screen_target[1] = np.clip(screen_target[1], 0, screen_h)
   437→   437→
   438→   438→                # Swipe detection
   439→   439→                swipe_gesture = swipe.update(filtered_pos, prev_pos, now)
   440→   440→                prev_pos = filtered_pos.copy()
   441→   441→
   442→   442→                if swipe_gesture == Gesture.SWIPE_LEFT:
   443→   443→                    _safe_kb_action(_kb(), "browser_back")
   444→   444→                    audio.click()
   445→   445→                elif swipe_gesture == Gesture.SWIPE_RIGHT:
   446→   446→                    _safe_kb_action(_kb(), "browser_forward")
   447→   447→                    audio.click()
   448→   448→
   449→   449→                # ═══ GESTURE ACTIONS ═══
   450→   450→
   451→   451→                # PINCH -> Left click
   452→   452→                if gesture == Gesture.PINCH and gesture_changed:
   453→   453→                    if now - last_click_time > config.pinch_cooldown:
   454→   454→                        mouse.left_click()
   455→   455→                        audio.click()
   456→   456→                        last_click_time = now
   457→   457→
   458→   458→                # PEACE -> Right click
   459→   459→                elif gesture == Gesture.PEACE and gesture_changed:
   460→   460→                    if now - last_click_time > config.pinch_cooldown:
   461→   461→                        mouse.right_click()
   462→   462→                        audio.right_click()
   463→   463→                        last_click_time = now
   464→   464→
   465→   465→                # THUMBS_UP -> Double click
   466→   466→                elif gesture == Gesture.THUMBS_UP and gesture_changed:
   467→   467→                    mouse.double_click()
   468→   468→                    audio.click()
   469→   469→                    last_click_time = now
   470→   470→
   471→   471→                # PINKY -> Middle click
   472→   472→                elif gesture == Gesture.PINKY and gesture_changed:
   473→   473→                    try:
   474→   474→                        mouse.mouse.click(mouse._button.middle, 1)
   475→   475→                    except Exception:
   476→   476→                        pass
   477→   477→                    audio.right_click()
   478→   478→                    last_click_time = now
   479→   479→
   480→   480→                # FIST -> Toggle freeze
   481→   481→                elif gesture == Gesture.FIST and gesture_changed:
   482→   482→                    cursor_frozen = not cursor_frozen
   483→   483→                    audio.freeze()
   484→   484→
   485→   485→                # OK -> Close window (Alt+F4)
   486→   486→                elif gesture == Gesture.OK and gesture_changed:
   487→   487→                    _safe_kb_action(_kb(), "close_window")
   488→   488→                    audio.click()
   489→   489→
   490→   490→                # SIX -> Task switcher (Alt+Tab)
   491→   491→                elif gesture == Gesture.SIX and gesture_changed:
   492→   492→                    _safe_kb_action(_kb(), "switch_window")
   493→   493→                    audio.click()
   494→   494→
   495→   495→                # PALM -> Drag mode
   496→   496→                if gesture == Gesture.PALM and not dragging:
   497→   497→                    mouse.start_drag()
   498→   498→                    dragging = True
   499→   499→                    audio.drag_start()
   500→   500→                elif gesture != Gesture.PALM and dragging:
   501→   501→                    mouse.stop_drag()
   502→   502→                    dragging = False
   503→   503→
   504→   504→                # THREE -> Scroll mode
   505→   505→                if gesture == Gesture.THREE:
   506→   506→                    scrolling = True
   507→   507→                    if gesture_changed:
   508→   508→                        prev_index_y = filtered_pos[1]
   509→   509→                        scroll_accum = 0.0
   510→   510→                    if prev_index_y is not None:
   511→   511→                        sd = (filtered_pos[1] - prev_index_y) * 40
   512→   512→                        scroll_accum += sd
   513→   513→                        if abs(scroll_accum) > 1.0:
   514→   514→                            mouse.scroll(int(scroll_accum))
   515→   515→                            audio.scroll_tick()
   516→   516→                            scroll_accum = 0.0
   517→   517→                        prev_index_y = filtered_pos[1]
   518→   518→                else:
   519→   519→                    scrolling = False
   520→   520→
   521→   521→                # GUN -> Snap to center
   522→   522→                if gesture == Gesture.GUN and gesture_changed:
   523→   523→                    spring.reset(center)
   524→   524→                    position_smoother.reset(center)
   525→   525→                    mouse.move_to(center[0], center[1])
   526→   526→                    home.reset()
   527→   527→                    audio.click()
   528→   528→
   529→   529→                # ROCK -> Minimize
   530→   530→                if gesture == Gesture.ROCK and gesture_changed:
   531→   531→                    _safe_kb_action(_kb(), "minimize_window")
   532→   532→                    audio.click()
   533→   533→
   534→   534→                # SHAKA -> Volume mode (move up/down to adjust)
   535→   535→                #   With debounce — don't spam volume changes
   536→   536→                if gesture == Gesture.SHAKA:
   537→   537→                    if not volume_mode and gesture_changed:
   538→   538→                        volume_mode = True
   539→   539→                        audio.mode_enter()
   540→   540→                    if gesture_changed:
   541→   541→                        prev_index_y = filtered_pos[1]
   542→   542→                    if prev_index_y is not None:
   543→   543→                        vd = filtered_pos[1] - prev_index_y
   544→   544→                        if abs(vd) > 0.02 and (now - last_volume_time > VOLUME_COOLDOWN):
   545→   545→                            k = _kb()
   546→   546→                            if k:
   547→   547→                                if vd > 0:
   548→   548→                                    _safe_kb_action(k, "volume_down")
   549→   549→                                else:
   550→   550→                                    _safe_kb_action(k, "volume_up")
   551→   551→                            prev_index_y = filtered_pos[1]
   552→   552→                            last_volume_time = now
   553→   553→                else:
   554→   554→                    if volume_mode:
   555→   555→                        volume_mode = False
   556→   556→                        audio.mode_exit()
   557→   557→
   558→   558→                # RING -> Brightness mode (move up/down to adjust)
   559→   559→                #   With debounce — brightness changes are slower
   560→   560→                if gesture == Gesture.RING:
   561→   561→                    if not brightness_mode and gesture_changed:
   562→   562→                        brightness_mode = True
   563→   563→                        audio.mode_enter()
   564→   564→                    if gesture_changed:
   565→   565→                        prev_index_y = filtered_pos[1]
   566→   566→                    if prev_index_y is not None:
   567→   567→                        bd = filtered_pos[1] - prev_index_y
   568→   568→                        if abs(bd) > 0.03 and (now - last_brightness_time > BRIGHTNESS_COOLDOWN):
   569→   569→                            k = _kb()
   570→   570→                            if k:
   571→   571→                                if bd > 0:
   572→   572→                                    _safe_kb_action(k, "brightness_down")
   573→   573→                                else:
   574→   574→                                    _safe_kb_action(k, "brightness_up")
   575→   575→                            prev_index_y = filtered_pos[1]
   576→   576→                            last_brightness_time = now
   577→   577→                else:
   578→   578→                    if brightness_mode:
   579→   579→                        brightness_mode = False
   580→   580→                        audio.mode_exit()
   581→   581→
   582→   582→                # ═══ CURSOR MOVEMENT ═══
   583→   583→                # Pointing and navigation gestures move cursor
   584→   584→                if gesture in (Gesture.POINTING, Gesture.PEACE,
   585→   585→                               Gesture.THUMBS_UP, Gesture.PINKY,
   586→   586→                               Gesture.GUN, Gesture.ROCK) and not cursor_frozen:
   587→   587→                    # Spring-damper physics
   588→   588→                    cursor_pos = spring.update(screen_target, dt)
   589→   589→                    # Edge gravity
   590→   590→                    cursor_pos += edge_grav.apply(cursor_pos, screen_w, screen_h)
   591→   591→                    # Momentum throw
   592→   592→                    cursor_pos += momentum.update(spring.velocity, True, dt)
   593→   593→                    # Clamp to screen
   594→   594→                    cursor_pos[0] = np.clip(cursor_pos[0], 0, screen_w)
   595→   595→                    cursor_pos[1] = np.clip(cursor_pos[1], 0, screen_h)
   596→   596→                    # Final position smoothing for silky output
   597→   597→                    smoothed_pos = position_smoother.smooth(cursor_pos)
   598→   598→                    mouse.move_to(smoothed_pos[0], smoothed_pos[1])
   599→   599→
   600→   600→                    speed = np.linalg.norm(spring.velocity)
   601→   601→                    if speed > 500:
   602→   602→                        audio.whoosh(speed)
   603→   603→
   604→   604→                elif cursor_frozen:
   605→   605→                    spring.update(spring.position, dt)
   606→   606→                    momentum.reset()
   607→   607→
   608→   608→                prev_gesture = gesture
   609→   609→
   610→   610→            else:
   611→   611→                # No hand detected
   612→   612→                hand_absent_frames += 1
   613→   613→
   614→   614→                # Momentum throw keeps cursor gliding
   615→   615→                throw = momentum.update(spring.velocity, False, dt)
   616→   616→                if momentum.is_active:
   617→   617→                    cp = spring.position + throw
   618→   618→                    cp[0] = np.clip(cp[0], 0, screen_w)
   619→   619→                    cp[1] = np.clip(cp[1], 0, screen_h)
   620→   620→                    smoothed = position_smoother.smooth(cp)
   621→   621→                    mouse.move_to(smoothed[0], smoothed[1])
   622→   622→
   623→   623→                spring.update(spring.position, dt)
   624→   624→
   625→   625→                # Only reset after hand has been absent for a few frames
   626→   626→                # This prevents jitter from momentary detection loss
   627→   627→                if hand_absent_frames > 5:
   628→   628→                    if not hand_was_lost:
   629→   629→                        audio.hand_lost()
   630→   630→                        hand_was_lost = True
   631→   631→                    jitter_x.reset()
   632→   632→                    jitter_y.reset()
   633→   633→                    home.reset()
   634→   634→                    velocity_predictor.reset()
   635→   635→                    gsm.reset()
   636→   636→                    prev_gesture = Gesture.NONE
   637→   637→                    prev_index_y = None
   638→   638→                    prev_pos = None
   639→   639→                    if dragging:
   640→   640→                        mouse.stop_drag()
   641→   641→                        dragging = False
   642→   642→                    scrolling = False
   643→   643→                    if volume_mode:
   644→   644→                        volume_mode = False
   645→   645→                    if brightness_mode:
   646→   646→                        brightness_mode = False
   647→   647→
   648→   648→            # Display
   649→   649→            if config.show_camera and debug_mode and hand_data["frame"] is not None:
   650→   650→                _draw_hud(hand_data["frame"], gesture_result, spring, fps,
   651→   651→                          config, cursor_frozen, dragging, scrolling,
   652→   652→                          volume_mode, brightness_mode, precision_mode,
   653→   653→                          gsm.progress, gesture_confidence)
   654→   654→                cv2.imshow("AirMouse", hand_data["frame"])
   655→   655→                key = cv2.waitKey(1) & 0xFF
   656→   656→                if key == ord("q"):
   657→   657→                    running = False
   658→   658→                elif key == ord("d"):
   659→   659→                    debug_mode = not debug_mode
   660→   660→                elif key == ord("r"):
   661→   661→                    home.reset()
   662→   662→                    spring.reset(center)
   663→   663→                    position_smoother.reset(center)
   664→   664→                    velocity_predictor.reset()
   665→   665→                    audio.recalibrate()
   666→   666→                    print("  -> Recalibrated")
   667→   667→                elif key == ord("s"):
   668→   668→                    config.audio_enabled = not config.audio_enabled
   669→   669→                    audio.enabled = config.audio_enabled
   670→   670→                    print(f"  -> Audio {'ON' if config.audio_enabled else 'OFF'}")
   671→   671→                elif key == ord("p"):
   672→   672→                    precision_mode = not precision_mode
   673→   673→                    audio.precision_toggle()
   674→   674→                    if precision_mode:
   675→   675→                        position_smoother = PositionSmoother(alpha=0.6)  # Smoother in precision mode
   676→   676→                    else:
   677→   677→                        position_smoother = PositionSmoother(alpha=config.position_smooth_alpha)
   678→   678→                    print(f"  -> Precision mode {'ON' if precision_mode else 'OFF'}"
   679→   679→                          f" (power={'1.0' if precision_mode else config.exp_power},"
   680→   680→                          f" scale={'1.0' if precision_mode else config.exp_scale})")
   681→   681→                elif key == ord("t"):
   682→   682→                    run_tutorial(tracker)
   683→   683→                elif key == ord("h"):
   684→   684→                    _show_quick_reference()
   685→   685→
   686→   686→            # FPS throttling
   687→   687→            elapsed = time.perf_counter() - t0
   688→   688→            frame_times.append(elapsed)
   689→   689→            if len(frame_times) > 30:
   690→   690→                frame_times.pop(0)
   691→   691→            avg = sum(frame_times) / len(frame_times)
   692→   692→            fps = 1.0 / max(avg, 1e-6)
   693→   693→            min_frame = 1.0 / config.target_fps
   694→   694→            if elapsed < min_frame:
   695→   695→                time.sleep(min_frame - elapsed)
   696→   696→
   697→   697→    except KeyboardInterrupt:
   698→   698→        print("\n  Shutting down...")
   699→   699→    finally:
   700→   700→        running = False
   701→   701→        tracker.release()
   702→   702→        if dragging:
   703→   703→            try:
   704→   704→                mouse.stop_drag()
   705→   705→            except Exception:
   706→   706→                pass
   707→   707→        try:
   708→   708→            if config.show_camera:
